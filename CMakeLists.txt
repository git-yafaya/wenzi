cmake_minimum_required(VERSION 3.16)
project(wenzi VERSION 0.1.0 LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(Threads REQUIRED)
include(CTest)

if(CMAKE_CROSSCOMPILING)
    set(_sherpa_default "${CMAKE_SYSROOT}/usr")
else()
    set(_sherpa_default "${CMAKE_CURRENT_SOURCE_DIR}/.deps/sherpa-patch-stage")
endif()
set(SHERPA_ONNX_ROOT "${_sherpa_default}" CACHE PATH "sherpa-onnx 的目标平台安装目录")
# 切换 SDK 前缀时重新查找，避免沿用另一平台的缓存。
unset(SHERPA_ONNX_INCLUDE_DIR CACHE)
unset(SHERPA_ONNX_C_API_LIBRARY CACHE)
find_path(SHERPA_ONNX_INCLUDE_DIR sherpa-onnx/c-api/c-api.h
    PATHS "${SHERPA_ONNX_ROOT}/include" NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)
find_library(SHERPA_ONNX_C_API_LIBRARY sherpa-onnx-c-api
    PATHS "${SHERPA_ONNX_ROOT}/lib" "${SHERPA_ONNX_ROOT}/lib64"
    NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)

if(NOT SHERPA_ONNX_INCLUDE_DIR OR NOT SHERPA_ONNX_C_API_LIBRARY)
    message(FATAL_ERROR "缺少 sherpa-onnx C API；请设置 SHERPA_ONNX_ROOT")
endif()
find_package(ALSA REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(LWS REQUIRED IMPORTED_TARGET libwebsockets>=4.3)
set(_backend_sources src/core.c src/asr_local.c src/punctuation.c
    src/asr_iflytek.c src/ws_server.c)
set(_backend_libraries "${SHERPA_ONNX_C_API_LIBRARY}" PkgConfig::LWS
    OpenSSL::SSL OpenSSL::Crypto Threads::Threads m)
add_executable(wenzi src/main.c src/audio_wav.c src/audio_alsa.c ${_backend_sources})
target_include_directories(wenzi PRIVATE include src "${SHERPA_ONNX_INCLUDE_DIR}")
target_compile_definitions(wenzi PRIVATE WENZI_WITH_NETWORK=1)
target_link_libraries(wenzi PRIVATE ${_backend_libraries} ALSA::ALSA)
set_target_properties(wenzi PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
install(TARGETS wenzi RUNTIME DESTINATION bin)

if(BUILD_TESTING)
    add_executable(test_audio tests/test_audio.c src/audio_wav.c)
    target_include_directories(test_audio PRIVATE include)
    target_compile_options(test_audio PRIVATE -UNDEBUG)
    add_test(NAME audio COMMAND test_audio)
    add_executable(test_network tests/test_network.c ${_backend_sources})
    target_include_directories(test_network PRIVATE include src "${SHERPA_ONNX_INCLUDE_DIR}")
    target_compile_definitions(test_network PRIVATE WENZI_WITH_NETWORK=1)
    target_compile_options(test_network PRIVATE -UNDEBUG)
    target_link_libraries(test_network PRIVATE ${_backend_libraries})
    add_test(NAME network COMMAND test_network)
    add_executable(test_core tests/test_core.c src/core.c)
    target_include_directories(test_core PRIVATE include src)
    target_compile_definitions(test_core PRIVATE WENZI_WITH_NETWORK=0)
    target_compile_options(test_core PRIVATE -UNDEBUG)
    add_test(NAME core COMMAND test_core)
endif()
